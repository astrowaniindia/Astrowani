# The Sitemap line below points at a BACKEND ROUTE, not a file in this directory.
# The blog agent publishes three articles a day into the `blogs` table and cannot
# commit to this git-deployed folder, so a static sitemap.xml here would be stale
# within hours. See astrowani-backend/src/blogSitemap.js.
#
# EVERY Disallow HERE MUST BE ABSENT FROM THAT SITEMAP, and vice versa.
# Submitting a URL in a sitemap while blocking it here is a self-inflicted
# Search Console error ("Indexed, though blocked by robots.txt"), and it is an
# easy one to create by accident: chat.html and call.html look like app screens
# but are in fact the "Chat with an Astrologer" / "Call an Astrologer" landing
# pages, which carry the highest-intent keywords on the site. They are indexed
# deliberately. Only genuinely per-user pages are blocked.
#
# WHEN THIS SITE MOVES TO astrowani.com: this file replaces the WordPress
# robots.txt generated by the ThinkRank SEO plugin. Its WordPress-only rules
# (/wp-admin/, /xmlrpc.php, /cart/, /checkout/, /my-account/) are deliberately
# NOT carried over — none of those paths exist here, and disallowing a path that
# does not exist only advertises what used to be there.

User-agent: *
Allow: /

# Per-user pages: behind a login, different for every visitor, nothing to index.
# Keeping crawlers out spends the crawl budget on the blog and the landing pages.
Disallow: /wallet.html

Sitemap: https://backend.astrowani.com/api/blogs/sitemap.xml
